# Story 6.5: Test Infrastructure

## Description

Провести комплексное тестирование всей Supabase инфраструктуры: подключение к БД, Google OAuth flow, RLS policies, Storage buckets. Создать тестовые скрипты для автоматизации проверок.

## Goals

- Протестировать все компоненты Supabase
- Создать тестовые скрипты для автоматизации
- Проверить Google OAuth flow end-to-end
- Проверить RLS policies для БД и Storage
- Проверить загрузку и скачивание файлов
- Задокументировать результаты тестирования

#### 5. RLS Policies Test (Database)

```sql
-- Create two test users
INSERT INTO auth.users (id, email, encrypted_password, email_confirmed_at, created_at, updated_at)
VALUES 
  ('user-1-id', 'user1@test.com', crypt('pass1', gen_salt('bf')), NOW(), NOW(), NOW()),
  ('user-2-id', 'user2@test.com', crypt('pass2', gen_salt('bf')), NOW(), NOW(), NOW());

-- Profiles should be created automatically via trigger
SELECT id FROM public.user_profiles WHERE id IN ('user-1-id', 'user-2-id');

-- Test as user-1: insert font
SET request.jwt.claim.sub = 'user-1-id';
INSERT INTO public.user_fonts (user_id, font_def)
VALUES ('user-1-id', '{"id": "test", "name": "Test Font", "source": "google"}'::jsonb);

-- Test as user-2: try to read user-1's fonts (should return 0 rows)
SET request.jwt.claim.sub = 'user-2-id';
SELECT * FROM public.user_fonts WHERE user_id = 'user-1-id';
-- Expected: 0 rows

-- Test as user-2: try to insert into user-1's data (should fail)
INSERT INTO public.user_fonts (user_id, font_def)
VALUES ('user-1-id', '{"id": "hack", "name": "Hack", "source": "google"}'::jsonb);
-- Expected: ERROR: new row violates row-level security policy

-- Cleanup
RESET request.jwt.claim.sub;
DELETE FROM auth.users WHERE id IN ('user-1-id', 'user-2-id');
```

#### Script 3: RLS Test (`test/rls-test.sql`)

```sql
-- RLS Test Script
BEGIN;

-- Create test users
INSERT INTO auth.users (id, email, encrypted_password, email_confirmed_at, created_at, updated_at)
VALUES 
  ('test-user-1', 'test1@example.com', crypt('pass', gen_salt('bf')), NOW(), NOW(), NOW()),
  ('test-user-2', 'test2@example.com', crypt('pass', gen_salt('bf')), NOW(), NOW(), NOW());

-- Wait for trigger to create profiles
DO $$ BEGIN PERFORM pg_sleep(0.1); END $$;

-- Test 1: User can see own profile
SET request.jwt.claim.sub = 'test-user-1';
SELECT CASE 
  WHEN COUNT(*) = 1 THEN 'PASS: User can see own profile'
  ELSE 'FAIL: User cannot see own profile'
END as test_result
FROM public.user_profiles WHERE id = 'test-user-1';

-- Test 2: User cannot see other's profile
SELECT CASE 
  WHEN COUNT(*) = 0 THEN 'PASS: User cannot see other profile'
  ELSE 'FAIL: User can see other profile (RLS breach!)'
END as test_result
FROM public.user_profiles WHERE id = 'test-user-2';

-- Test 3: User can insert own font
INSERT INTO public.user_fonts (user_id, font_def)
VALUES ('test-user-1', '{"id": "test", "name": "Test", "source": "google"}'::jsonb);

SELECT CASE 
  WHEN COUNT(*) = 1 THEN 'PASS: User can insert own font'
  ELSE 'FAIL: User cannot insert own font'
END as test_result
FROM public.user_fonts WHERE user_id = 'test-user-1';

-- Test 4: User cannot insert font for another user
DO $$
BEGIN
  INSERT INTO public.user_fonts (user_id, font_def)
  VALUES ('test-user-2', '{"id": "hack", "name": "Hack", "source": "google"}'::jsonb);
  RAISE EXCEPTION 'FAIL: User can insert font for another user (RLS breach!)';
EXCEPTION
  WHEN OTHERS THEN
    RAISE NOTICE 'PASS: User cannot insert font for another user';
END $$;

-- Cleanup
ROLLBACK;
```

#### Script 4: Storage Test (`test/storage-test.sh`)

```bash
#!/bin/bash

echo "=== Storage Test ==="

# Variables (set these)
ACCESS_TOKEN="${ACCESS_TOKEN:-}"
USER_ID="${USER_ID:-}"

if [ -z "$ACCESS_TOKEN" ] || [ -z "$USER_ID" ]; then
  echo "ERROR: Set ACCESS_TOKEN and USER_ID environment variables"
  exit 1
fi

# Create test files
echo "Creating test files..."
echo "Test font content" > test-font.woff2
echo "Test image content" > test-image.jpg

# Test 1: Upload font
echo "Test 1: Upload font..."
RESPONSE=$(curl -s -w "\n%{http_code}" -X POST \
  "http://localhost:8000/storage/v1/object/user-fonts/${USER_ID}/test-font.woff2" \
  -H "Authorization: Bearer ${ACCESS_TOKEN}" \
  -H "Content-Type: font/woff2" \
  --data-binary @test-font.woff2)

HTTP_CODE=$(echo "$RESPONSE" | tail -n1)
if [ "$HTTP_CODE" = "200" ]; then
  echo "✓ PASS: Font uploaded"
else
  echo "✗ FAIL: Font upload failed (HTTP $HTTP_CODE)"
fi

# Test 2: Download font
echo "Test 2: Download font..."
HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" \
  "http://localhost:8000/storage/v1/object/public/user-fonts/${USER_ID}/test-font.woff2")

if [ "$HTTP_CODE" = "200" ]; then
  echo "✓ PASS: Font downloaded"
else
  echo "✗ FAIL: Font download failed (HTTP $HTTP_CODE)"
fi

# Test 3: Try to upload to another user's folder
echo "Test 3: Try to upload to another user's folder..."
HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" -X POST \
  "http://localhost:8000/storage/v1/object/user-fonts/other-user-id/hack.woff2" \
  -H "Authorization: Bearer ${ACCESS_TOKEN}" \
  -H "Content-Type: font/woff2" \
  --data-binary @test-font.woff2)

if [ "$HTTP_CODE" != "200" ]; then
  echo "✓ PASS: Upload to other folder blocked (HTTP $HTTP_CODE)"
else
  echo "✗ FAIL: Upload to other folder allowed (RLS breach!)"
fi

# Cleanup
rm -f test-font.woff2 test-image.jpg

echo "=== Storage Test Complete ==="
```
